# promo-web 运营推广前端 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 创建 React + Vite + Tailwind SPA 前端(活动列表 + 数据看板),调用已构建的运营推广 API,并发布到 nubase Assets CDN。

**Architecture:** 独立应用 `frontend/apps/promo-web`(pnpm workspace 成员)。两个路由页面通过统一 API client(anon/authenticated 项目 token)调用 `/rest/v1/*`。构建产物经 `assetsUpload` 上传到 `/assets/v1/promo/`,`assetsUpdateSettings` 设置 spaFallback 支持前端路由。

**Tech Stack:** React 18 / Vite 6 / TypeScript 5.5 / Tailwind CSS 3.4 / react-router-dom 6 / recharts 2 / pnpm 10

**Spec:** `docs/superpowers/specs/2026-08-13-promo-web-design.md`

```typescript
import type { Campaign, CampaignCount, DashboardSummary, TrendPoint } from '../types';

const ANON_KEY = import.meta.env.VITE_NUBASE_ANON_KEY as string | undefined;

if (!ANON_KEY) {
  console.warn('VITE_NUBASE_ANON_KEY is not set — API calls will fail');
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, {
    ...init,
    headers: {
      apikey: ANON_KEY ?? '',
      Authorization: `Bearer ${ANON_KEY ?? ''}`,
      'Content-Type': 'application/json',
      ...init?.headers,
    },
  });
  if (!res.ok) {
    throw new Error(`API ${res.status}: ${await res.text()}`);
  }
  return res.json() as Promise<T>;
}

export const api = {
  listCampaigns: () =>
    request<Campaign[]>(
      '/rest/v1/campaigns?select=id,name,description,status,start_time,end_time,max_participants&order=start_time.desc'
    ),
  dashboardSummary: () =>
    request<[{ result: DashboardSummary }]>('/rest/v1/rpc/dashboard_summary', {
      method: 'POST',
      body: '{}',
    }),
  participationTrend: () =>
    request<TrendPoint[]>('/rest/v1/rpc/participation_trend', {
      method: 'POST',
      body: '{}',
    }),
  campaignCounts: () =>
    request<CampaignCount[]>('/rest/v1/rpc/campaign_participation_counts', {
      method: 'POST',
      body: '{}',
    }),
};
```

import type { Campaign } from '../types';
import StatusBadge from './StatusBadge';

function formatDate(iso: string | null): string {
  if (!iso) return '长期';
  return new Date(iso).toLocaleDateString('zh-CN', { month: '2-digit', day: '2-digit' });
}

export default function CampaignCard({
  campaign,
  participationCount,
}: {
  campaign: Campaign;
  participationCount: number;
}) {
  const progress =
    campaign.max_participants != null
      ? Math.min(100, Math.round((participationCount / campaign.max_participants) * 100))
      : null;

  return (
    <div className="bg-white rounded-xl border border-gray-200 p-5 hover:shadow-md transition-shadow flex flex-col">
      <div className="flex items-start justify-between gap-3">
        <h3 className="text-base font-semibold text-gray-900">{campaign.name}</h3>
        <StatusBadge status={campaign.status} />
      </div>
      {campaign.description && (
        <p className="mt-2 text-sm text-gray-500 line-clamp-2 flex-1">{campaign.description}</p>
      )}
      <p className="mt-3 text-xs text-gray-400">
        {formatDate(campaign.start_time)} ~ {formatDate(campaign.end_time)}
      </p>
      {campaign.max_participants != null && (
        <div className="mt-3">
          <div className="flex justify-between text-xs text-gray-500 mb-1">
            <span>参与进度</span>
            <span>
              {participationCount}/{campaign.max_participants}
            </span>
          </div>
          <div className="h-1.5 bg-gray-100 rounded-full overflow-hidden">
            <div
              className="h-full bg-indigo-500 rounded-full"
              style={{ width: `${progress}%` }}
            />
          </div>
        </div>
      )}
    </div>
  );
}

### Task 5: 活动列表页

**Files:**

- [ ] **Step 1: 创建页面**

```tsx
import { useCallback, useEffect, useState } from 'react';
import { api } from '../lib/api';
import type { Campaign, CampaignStatus } from '../types';
import CampaignCard from '../components/CampaignCard';

type Filter = 'all' | CampaignStatus;

export default function CampaignListPage() {
  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
  const [counts, setCounts] = useState<Record<number, number>>({});
  const [filter, setFilter] = useState<Filter>('all');
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const [list, cnt] = await Promise.all([api.listCampaigns(), api.campaignCounts()]);
      setCampaigns(list);
      setCounts(Object.fromEntries(cnt.map((c) => [c.campaign_id, c.participation_count])));
    } catch (e) {
      setError(e instanceof Error ? e.message : '加载失败');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  const visible = filter === 'all' ? campaigns : campaigns.filter((c) => c.status === filter);

  const filters: [Filter, string][] = [
    ['all', '全部'],
    ['active', '进行中'],
    ['closed', '已结束'],
  ];

  return (
    <div>
      <div className="flex items-center justify-between mb-5">
        <h2 className="text-xl font-bold text-gray-900">活动列表</h2>
        <div className="flex gap-2">
          {filters.map(([value, label]) => (
            <button
              key={value}
              onClick={() => setFilter(value)}
              className={`px-3 py-1.5 rounded-lg text-sm font-medium transition-colors ${
                filter === value
                  ? 'bg-indigo-600 text-white'
                  : 'bg-white text-gray-600 border border-gray-200 hover:bg-gray-50'
              }`}
            >
              {label}
            </button>
          ))}
        </div>
      </div>

      {loading && <p className="text-gray-500 py-10 text-center">加载中...</p>}

      {error && (
        <div className="bg-red-50 border border-red-200 rounded-xl p-4 text-sm text-red-600">
          <p>{error}</p>
          <button onClick={() => void load()} className="mt-2 text-indigo-600 hover:underline">
            重试
          </button>
        </div>
      )}

      {!loading && !error && visible.length === 0 && (
        <p className="text-gray-500 py-10 text-center">暂无活动</p>
      )}

      {!loading && !error && visible.length > 0 && (
        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
          {visible.map((c) => (
            <CampaignCard key={c.id} campaign={c} participationCount={counts[c.id] ?? 0} />
          ))}
        </div>
      )}
    </div>
  );
}
```

import { useCallback, useEffect, useState } from 'react';
import {
  Bar,
  BarChart,
  CartesianGrid,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from 'recharts';
import { api } from '../lib/api';
import type { CampaignCount, DashboardSummary, TrendPoint } from '../types';
import StatCard from '../components/StatCard';
import StatusBadge from '../components/StatusBadge';

export default function DashboardPage() {
  const [summary, setSummary] = useState<DashboardSummary | null>(null);
  const [trend, setTrend] = useState<TrendPoint[]>([]);
  const [counts, setCounts] = useState<CampaignCount[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const [s, t, c] = await Promise.all([
        api.dashboardSummary(),
        api.participationTrend(),
        api.campaignCounts(),
      ]);
      setSummary(s[0]?.result ?? null);
      setTrend(t);
      setCounts(c);
    } catch (e) {
      setError(e instanceof Error ? e.message : '加载失败');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  const chartData = trend.map((p) => ({ day: p.day.slice(5), count: p.count }));

  return (
    <div>
      <div className="flex items-center justify-between mb-5">
        <h2 className="text-xl font-bold text-gray-900">数据看板</h2>
      </div>

      {loading && <p className="text-gray-500 py-10 text-center">加载中...</p>}

      {error && (
        <div className="bg-red-50 border border-red-200 rounded-xl p-4 text-sm text-red-600">
          <p>{error}</p>
          <button onClick={() => void load()} className="mt-2 text-indigo-600 hover:underline">
            重试
          </button>
        </div>
      )}

      {!loading && !error && summary && (
        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4 mb-6">
          <StatCard label="活动总数" value={summary.total_campaigns} />
          <StatCard label="进行中活动" value={summary.active_campaigns} accent="text-green-600" />

          <StatCard label="今日参与" value={summary.today_participations} accent="text-orange-600" />
        </div>
      )}

      {!loading && !error && chartData.length > 0 && (
        <div className="bg-white rounded-xl border border-gray-200 p-5 mb-6">
          <h3 className="text-sm font-semibold text-gray-700 mb-4">参与趋势(按日)</h3>

                <CartesianGrid strokeDasharray="3 3" vertical={false} />
                <XAxis dataKey="day" tick={{ fontSize: 12 }} />
                <YAxis allowDecimals={false} tick={{ fontSize: 12 }} />
                <Tooltip />
                <Bar dataKey="count" name="参与人数" fill="#4f46e5" radius={[4, 4, 0, 0]} />

            </ResponsiveContainer>
          </div>
        </div>

        <div className="bg-white rounded-xl border border-gray-200 p-5">

                      : '—'}
